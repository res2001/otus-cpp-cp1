17 typename = std::enable_if_t<(Dim > 0 && Dim <= MatrixMaxDimension)>>
18struct MatrixIndex

20 static constexpr size_t BAD_IDX = SIZE_MAX;

27 assert(list.size() == Dim && "Количество индексов не совпадает с размерностью матрицы!");
28 std::copy(list.begin(), list.end(), index_arr.begin());

38 ss << index_arr[Dim - 1] << "]";
39 return ss.str();
40 }

43 assert(dim_idx < Dim);
44 index_arr[dim_idx] = idx;
